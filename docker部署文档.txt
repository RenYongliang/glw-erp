[mysqld]
# ============================================================
# 场景：20 人以内自建系统，35 个表，最大表约 10 万行
# 目标：低内存占用、稳定运行，把内存让给 Java 后端
# 适用：MySQL 8.0，Docker 容器内存限制建议 600M
# ============================================================

# ---------- 核心缓冲池 ----------
# 默认值：128M（系统内存 < 1GB 时）
# 调整原因：你的全库数据和索引远小于 256M，256M 可缓存全部热点数据，
#           基本消除读磁盘 IO；同时不会挤占 Java 后端内存。
innodb_buffer_pool_size = 256M

# 默认值：1（buffer pool < 1GB 时）
# 保持默认：单实例即可，小内存下多实例无收益。
innodb_buffer_pool_instances = 1

# 默认值：16M
# 保持默认：写入压力小，16M 日志缓冲足够。
innodb_log_buffer_size = 16M

# ---------- 关闭高内存开销的诊断功能 ----------
# 默认值：ON
# 调整原因：Performance Schema 默认开启，会消耗约 100-400MB 内存。
#           对 20 人以内、无专职 DBA 的场景，关闭它几乎没有负面影响，
#           是小内存机器上性价比最高的优化。
performance_schema = OFF

# ---------- 连接数控制 ----------
# 默认值：151
# 调整原因：20 人以内系统，同时活跃连接通常个位数。
#           限制为 20，并与 Java 连接池（建议 maximumPoolSize=10）匹配，
#           彻底切断因连接数虚高导致会话内存成倍占用的风险。
max_connections = 20

# 默认值：28800 秒（8 小时）
# 调整原因：默认 8 小时太长，空闲连接会一直占用资源。
#           改为 60 秒，闲置连接 1 分钟后自动断开，快速释放会话内存。
wait_timeout = 60
interactive_timeout = 60

# ---------- 每连接缓冲（保持保守默认值）----------
# 默认值：256K
# 保持默认：每连接排序缓冲，20 人场景默认值完全够用。
sort_buffer_size = 256K

# 默认值：256K
# 保持默认：无索引 JOIN 时的缓冲。你已建索引，默认值足够。
join_buffer_size = 256K

# 默认值：128K
# 保持默认：顺序扫描缓冲，默认值合理。
read_buffer_size = 128K

# 默认值：256K
# 保持默认：随机读取缓冲，默认值合理。
read_rnd_buffer_size = 256K

# ---------- 临时表上限 ----------
# 默认值：16M
# 调整原因：适度提升到 32M，为复杂 GROUP BY / JOIN 留余量，
#           确保 10 万行级别的中间结果在内存中完成，避免写磁盘。
tmp_table_size = 32M
max_heap_table_size = 32M

# ---------- 慢查询日志（建议开启）----------
# 默认值：OFF
# 调整原因：20 人场景下，任何超过 1 秒的查询都值得排查。
#           数据量小，慢查询日志开销可忽略。
slow_query_log = ON
slow_query_log_file = /var/lib/mysql/slow.log
long_query_time = 1


sudo docker update --memory=600m --memory-swap=900m mysql
sudo docker update --memory=128m --memory-swap=256m redis
sudo docker update --memory=256m --memory-swap=512m silo
sudo docker update --memory=1g --memory-swap=1280m glw-backend
sudo docker update --memory=128m --memory-swap=256m glw-web


# 构建后端镜像
sudo docker build -t glw-backend:1.0.0 .

# 构建前端镜像
sudo docker build -t glw-web:1.0.0 .


# 运行mysql
docker run -d \
  --name mysql \
  --network glw-network \
  -p 3306:3306 \
  -e MYSQL_ROOT_PASSWORD='ryl123' \
  -e TZ=Asia/Shanghai \
  -v /data/mysql/data:/var/lib/mysql \
  mysql:8.4


# 运行redis
docker run -d \
  --name redis \
  --network glw-network \
  -p 6379:6379 \
  -v /data/redis/data:/data \
  redis:8.2 \
  redis-server --requirepass 'ryl123'


# 运行silo
docker run -d \
  --name silo \
  --network glw-network \
  -p 9000:9000 \
  -p 9001:9001 \
  -e MINIO_ROOT_USER=root \
  -e MINIO_ROOT_PASSWORD='3306ryl^Coder' \
  -v /data/silo/data:/data \
  docker.io/pgsty/silo:RELEASE.2026-11-12T00-00-00Z \
  server /data --console-address ":9001"


# 运行后端
sudo docker run -d \
  --name glw-backend \
  --network glw-network \
  -e SPRING_PROFILES_ACTIVE=dev \
  -e DB_URL='jdbc:mysql://mysql:3306/glw?useUnicode=true&characterEncoding=UTF-8&serverTimezone=Asia/Shanghai' \
  -e REDIS_HOST=redis \
  -e MINIO_ENDPOINT=http://minio:9000 \
  -e JAVA_TOOL_OPTIONS="-Xms256m -Xmx768m" \
  -e TZ=Asia/Shanghai \
  glw-backend:1.0.0

# 运行前端
sudo docker run -d \
  --name glw-web \
  --network glw-network \
  -p 80:80 \
  -p 443:443 \
  glw-web:1.0.0
