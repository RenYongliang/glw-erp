[mysqld]
# ============================================================
# 场景：20 人以内自建系统，35 个表，最大表约 10 万行
# 目标：低内存占用、稳定运行，把内存让给 Java 后端
# 适用：MySQL 8.0，Docker 容器内存限制建议 600M
# ============================================================

# ---------- 核心缓冲池 ----------
# 默认值：128M（系统内存 < 1GB 时）
# 调整原因：你的全库数据和索引远小于 256M，256M 可缓存全部热点数据，
#           基本消除读磁盘 IO；同时不会挤占 Java 后端内存。
innodb_buffer_pool_size = 256M

# 默认值：1（buffer pool < 1GB 时）
# 保持默认：单实例即可，小内存下多实例无收益。
innodb_buffer_pool_instances = 1

# 默认值：16M
# 保持默认：写入压力小，16M 日志缓冲足够。
innodb_log_buffer_size = 16M

# ---------- 关闭高内存开销的诊断功能 ----------
# 默认值：ON
# 调整原因：Performance Schema 默认开启，会消耗约 100-400MB 内存。
#           对 20 人以内、无专职 DBA 的场景，关闭它几乎没有负面影响，
#           是小内存机器上性价比最高的优化。
performance_schema = OFF

# ---------- 连接数控制 ----------
# 默认值：151
# 调整原因：20 人以内系统，同时活跃连接通常个位数。
#           限制为 20，并与 Java 连接池（建议 maximumPoolSize=10）匹配，
#           彻底切断因连接数虚高导致会话内存成倍占用的风险。
max_connections = 20

# 默认值：28800 秒（8 小时）
# 调整原因：默认 8 小时太长，空闲连接会一直占用资源。
#           改为 1800 秒，闲置连接 30 分钟后自动断开，快速释放会话内存。
wait_timeout=1800
interactive_timeout=1800

# ---------- 每连接缓冲（保持保守默认值）----------
# 默认值：256K
# 保持默认：每连接排序缓冲，20 人场景默认值完全够用。
sort_buffer_size = 256K

# 默认值：256K
# 保持默认：无索引 JOIN 时的缓冲。你已建索引，默认值足够。
join_buffer_size = 256K

# 默认值：128K
# 保持默认：顺序扫描缓冲，默认值合理。
read_buffer_size = 128K

# 默认值：256K
# 保持默认：随机读取缓冲，默认值合理。
read_rnd_buffer_size = 256K

# ---------- 临时表上限 ----------
# 默认值：16M
# 调整原因：适度提升到 32M，为复杂 GROUP BY / JOIN 留余量，
#           确保 10 万行级别的中间结果在内存中完成，避免写磁盘。
tmp_table_size = 32M
max_heap_table_size = 32M

# ---------- 慢查询日志（建议开启）----------
# 默认值：OFF
# 调整原因：20 人场景下，任何超过 1 秒的查询都值得排查。
#           数据量小，慢查询日志开销可忽略。
slow_query_log = ON
slow_query_log_file = /var/lib/mysql/slow.log
long_query_time = 1

SSH 隧道是把服务器上的端口，通过加密 SSH 连接映射到你本地：

ssh -L 9001:127.0.0.1:9001 originalme@43.142.173.92

执行后，在你 Windows 浏览器打开：

http://localhost:9001

实际访问的是服务器上的 MinIO 控制台，但：

- 不需要把 9001 暴露到公网
- 只有你本人通过 SSH 登录后才能访问
- 流量走 SSH 加密

它不是"对外开放"，而是私有访问。

前提是服务器上的 9001 至少监听在 127.0.0.1。Compose 里可以这样：

silo:
  ports:
    - "127.0.0.1:9001:9001"



2. 怎么删除旧网络

你现在的 compose 已经改成：

networks:
  glw-network:
    name: glw-network

这个网络会由 Compose 创建和管理。旧网络是手工创建的，缺少 Compose 标签，直接启动可能报：

network glw-network was found but has incorrect label ...




隧道映射正确顺序：

cd /opt/project/glw

# 1. 查看网络里还有哪些容器
docker network inspect glw-network --format '{{range .Containers}}{{.Name}} {{end}}'

# 2. 删除旧容器（数据在 /data，不会丢）
docker rm -f mysql redis silo glw-backend glw-web

# 3. 删除旧网络
docker network rm glw-network

# 4. 用 compose 创建新网络并启动
docker compose --env-file envs/dev.env up -d

如果第 3 步报 network has active endpoints，说明还有其他容器连着它，先停掉或断开那些容器。

正确做法：本地端口和服务器端口错开

Compose 里服务器侧仍然只绑定回环：

  mysql:
    ports:
      - "127.0.0.1:3306:3306"

  redis:
    ports:
      - "127.0.0.1:6379:6379"

  silo:
    ports:
      - "127.0.0.1:9000:9000"
      - "127.0.0.1:9001:9001"

本地建隧道时映射到不同端口，例如：

ssh -N \
  -L 13306:127.0.0.1:3306 \
  -L 16379:127.0.0.1:6379 \
  -L 19000:127.0.0.1:9000 \
  -L 19001:127.0.0.1:9001 \
  originalme@43.142.173.92

含义：

本地 13306 → 服务器的 127.0.0.1:3306（MySQL）
本地 16379 → 服务器的 127.0.0.1:6379（Redis）
本地 19001 → 服务器的 127.0.0.1:9001（MinIO 控制台）
